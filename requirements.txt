# ─── Core framework and server ───────────────────────────────────────────────
# uvicorn[standard] is required, not just uvicorn: the WebSocket support
# for /ws/flights comes from the `websockets` package, which the bare
# `uvicorn` distribution does NOT include. Without it uvicorn answers
# "Unsupported upgrade request" and every live flight feed silently 404s.
fastapi==0.141.1
uvicorn[standard]==0.37.0
websockets==15.0.1
pydantic==2.13.5
python-dotenv==1.2.1

# ─── Database ────────────────────────────────────────────────────────────────
SQLAlchemy==2.1.1

# PostgreSQL driver — uncomment when moving off SQLite.
#   pip install alembic psycopg[binary]
# alembic==1.16.4
# psycopg[binary]==3.2.9

# ─── HTTP client for OpenSky ────────────────────────────────────────────────
# Already pulled in by uvicorn[standard]; listed explicitly because
# OpenSkyService depends on it directly.
httpx==0.28.1

# ─── Testing ─────────────────────────────────────────────────────────────────
pytest==8.4.2
